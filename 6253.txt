WEEK-3 ROUND ROBIN PROGRAM

#include <stdio.h>

int main() {
    int n, tq, i, rem[10], bt[10];
    int time = 0, done;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    printf("Enter burst times:\n");
    for(i = 0; i < n; i++) {
        scanf("%d", &bt[i]);
        rem[i] = bt[i];
    }

    printf("Enter Time Quantum: ");
    scanf("%d", &tq);

    do {
        done = 1;

        for(i = 0; i < n; i++) {
            if(rem[i] > 0) {
                done = 0;

                if(rem[i] > tq) {
                    time += tq;
                    rem[i] -= tq;
                } else {
                    time += rem[i];
                    rem[i] = 0;
                }
            }
        }
    } while(!done);

    printf("Total Execution Time = %d\n", time);

    return 0;
}


PRIORITY SCHEDULING


#include <stdio.h>

int main() {
    int n, i, j, time = 0, completed = 0;
    int at[10], bt[10], pr[10];
    int ct[10], tat[10], wt[10], done[10];

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        printf("\nProcess P%d\n", i + 1);

        printf("Arrival Time: ");
        scanf("%d", &at[i]);

        printf("Burst Time: ");
        scanf("%d", &bt[i]);

        printf("Priority: ");
        scanf("%d", &pr[i]);

        done[i] = 0;
    }

    while(completed < n) {
        int highest = -1;

        for(i = 0; i < n; i++) {
            if(at[i] <= time && done[i] == 0) {
                if(highest == -1 || pr[i] < pr[highest])
                    highest = i;
            }
        }

        if(highest == -1) {
            time++;
        } else {
            time += bt[highest];
            ct[highest] = time;
            tat[highest] = ct[highest] - at[highest];
            wt[highest] = tat[highest] - bt[highest];
            done[highest] = 1;
            completed++;
        }
    }

    printf("\nP\tAT\tBT\tPR\tCT\tTAT\tWT\n");

    for(i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\t%d\t%d\t%d\n",
               i + 1, at[i], bt[i], pr[i],
               ct[i], tat[i], wt[i]);
    }

    return 0;
}


WEEK -8 SEGMENTATION MEMORY


#include <stdio.h>

int main() {
    int n, i;
    int base[10], limit[10];
    int segno, offset, physical;

    printf("Enter number of segments: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        printf("\nSegment %d\n", i);
        printf("Base Address: ");
        scanf("%d", &base[i]);

        printf("Limit: ");
        scanf("%d", &limit[i]);
    }

    printf("\nEnter Segment Number: ");
    scanf("%d", &segno);

    printf("Enter Offset: ");
    scanf("%d", &offset);

    if(offset < limit[segno]) {
        physical = base[segno] + offset;
        printf("Physical Address = %d", physical);
    }
    else {
        printf("Segmentation Fault! Invalid Offset");
    }

    return 0;
}


ROUND ROBIN USING ARRIVAL TIMES


#include <stdio.h>

int main() {
    int n, tq, time = 0, remain, i;
    int at[10], bt[10], rt[10];
    int ct[10], tat[10], wt[10];

    printf("Enter number of processes: ");
    scanf("%d", &n);

    remain = n;

    for(i = 0; i < n; i++) {
        printf("\nProcess P%d\n", i + 1);

        printf("Arrival Time: ");
        scanf("%d", &at[i]);

        printf("Burst Time: ");
        scanf("%d", &bt[i]);

        rt[i] = bt[i];
    }

    printf("\nEnter Time Quantum: ");
    scanf("%d", &tq);

    while(remain > 0) {
        int executed = 0;

        for(i = 0; i < n; i++) {
            if(at[i] <= time && rt[i] > 0) {
                executed = 1;

                if(rt[i] > tq) {
                    time += tq;
                    rt[i] -= tq;
                } else {
                    time += rt[i];
                    ct[i] = time;
                    rt[i] = 0;
                    remain--;
                }
            }
        }

        if(executed == 0)
            time++;
    }

    printf("\nP\tAT\tBT\tCT\tTAT\tWT\n");

    for(i = 0; i < n; i++) {
        tat[i] = ct[i] - at[i];
        wt[i] = tat[i] - bt[i];

        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               i + 1, at[i], bt[i],
               ct[i], tat[i], wt[i]);
    }

    return 0;
}
