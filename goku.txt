#include<stdio.h>

int main()
{
    int n,i;
    int at[10],bt[10],pr[10];
    int wt[10],tat[10],completed[10]={0};

    int time=0,count=0;

    printf("Enter number of processes: ");
    scanf("%d",&n);

    for(i=0;i<n;i++)
    {
        printf("Enter AT BT Priority for P%d: ",i+1);
        scanf("%d%d%d",&at[i],&bt[i],&pr[i]);
    }

    while(count<n)
    {
        int pos=-1;

        for(i=0;i<n;i++)
        {
            if(at[i]<=time && completed[i]==0)
            {
                if(pos==-1 || pr[i]<pr[pos])
                    pos=i;
            }
        }

        if(pos==-1)
        {
            time++;
        }
        else
        {
            wt[pos]=time-at[pos];
            time=time+bt[pos];
            tat[pos]=wt[pos]+bt[pos];

            completed[pos]=1;
            count++;
        }
    }

    printf("\nProcess\tWT\tTAT\n");

    for(i=0;i<n;i++)
        printf("P%d\t%d\t%d\n",i+1,wt[i],tat[i]);

    return 0;
}

#include<stdio.h>

int main()
{
    int ms,ps,nop,np,rempages;
    int i,j,x,y,pa,offset;
    int s[10],fno[10][20];

    printf("Enter memory size: ");
    scanf("%d",&ms);

    printf("Enter page size: ");
    scanf("%d",&ps);

    nop=ms/ps;

    printf("Number of pages available = %d\n",nop);

    printf("Enter number of processes: ");
    scanf("%d",&np);

    rempages=nop;

    for(i=0;i<np;i++)
    {
        printf("Enter number of pages required for P%d: ",i);
        scanf("%d",&s[i]);

        if(s[i]>rempages)
        {
            printf("Memory is Full\n");
            break;
        }

        rempages=rempages-s[i];

        printf("Enter page table for P%d:\n",i);

        for(j=0;j<s[i];j++)
            scanf("%d",&fno[i][j]);
    }

    printf("\nEnter Process No, Page No and Offset: ");
    scanf("%d%d%d",&x,&y,&offset);

    if(x>=np || y>=s[x] || offset>=ps)
    {
        printf("Invalid Process/Page Number/Offset");
    }
    else
    {
        pa=fno[x][y]*ps+offset;
        printf("Physical Address = %d",pa);
    }

    return 0;
}





#include<stdio.h>

int main()
{
    int n,i,time=0,remain,tq;
    int at[10],bt[10],rt[10];
    int wt[10]={0},tat[10];

    printf("Enter number of processes: ");
    scanf("%d",&n);

    for(i=0;i<n;i++)
    {
        printf("Enter AT and BT for P%d: ",i+1);
        scanf("%d%d",&at[i],&bt[i]);
        rt[i]=bt[i];
    }

    printf("Enter Time Quantum: ");
    scanf("%d",&tq);

    remain=n;
    i=0;

    while(remain!=0)
    {
        if(rt[i]>0 && at[i]<=time)
        {
            if(rt[i]<=tq)
            {
                time=time+rt[i];
                rt[i]=0;

                tat[i]=time-at[i];
                wt[i]=tat[i]-bt[i];

                remain--;
            }
            else
            {
                rt[i]=rt[i]-tq;
                time=time+tq;
            }
        }
        else if(at[i]>time)
        {
            time++;
        }

        i=(i+1)%n;
    }

    printf("\nProcess\tWT\tTAT\n");

    for(i=0;i<n;i++)
        printf("P%d\t%d\t%d\n",i+1,wt[i],tat[i]);

    return 0;
}


#include<stdio.h>

int main()
{
    int n,i,j;
    int bt[10],pr[10],pid[10];
    int wt[10],tat[10];

    printf("Enter number of processes: ");
    scanf("%d",&n);

    for(i=0;i<n;i++)
    {
        pid[i]=i+1;
        printf("Enter BT and Priority for P%d: ",i+1);
        scanf("%d%d",&bt[i],&pr[i]);
    }

    for(i=0;i<n-1;i++)
    {
        for(j=i+1;j<n;j++)
        {
            if(pr[i]>pr[j])
            {
                int temp;

                temp=pr[i];
                pr[i]=pr[j];
                pr[j]=temp;

                temp=bt[i];
                bt[i]=bt[j];
                bt[j]=temp;

                temp=pid[i];
                pid[i]=pid[j];
                pid[j]=temp;
            }
        }
    }

    wt[0]=0;

    for(i=1;i<n;i++)
        wt[i]=wt[i-1]+bt[i-1];

    for(i=0;i<n;i++)
        tat[i]=wt[i]+bt[i];

    printf("\nPID\tWT\tTAT\n");

    for(i=0;i<n;i++)
        printf("P%d\t%d\t%d\n",pid[i],wt[i],tat[i]);

    return 0;
}
