#include <stdio.h>

int main() {
    int n, bt[10], wt[10], tat[10], i, j, temp;

    printf("Enter no of processes: ");
    scanf("%d", &n);

    for(i=0;i<n;i++){
        printf("BT of P%d: ", i+1);
        scanf("%d", &bt[i]);
    }

    for(i=0;i<n-1;i++)
        for(j=i+1;j<n;j++)
            if(bt[i] > bt[j]){
                temp=bt[i];
                bt[i]=bt[j];
                bt[j]=temp;
            }

    wt[0]=0;
    for(i=1;i<n;i++)
        wt[i]=wt[i-1]+bt[i-1];

    for(i=0;i<n;i++){
        tat[i]=wt[i]+bt[i];
        printf("P%d WT=%d TAT=%d\n", i+1, wt[i], tat[i]);
    }
    return 0;
}



#include <stdio.h>

int main() {
    int n, at[10], bt[10], rt[10], tq, time=0, done;
    
    printf("No of processes: ");
    scanf("%d",&n);

    for(int i=0;i<n;i++){
        scanf("%d%d",&at[i],&bt[i]);
        rt[i]=bt[i];
    }

    printf("Time Quantum: ");
    scanf("%d",&tq);

    do{
        done=1;
        for(int i=0;i<n;i++){
            if(at[i]<=time && rt[i]>0){
                done=0;
                if(rt[i]>tq){
                    time+=tq;
                    rt[i]-=tq;
                }else{
                    time+=rt[i];
                    printf("P%d completed at %d\n",i+1,time);
                    rt[i]=0;
                }
            }
        }
    }while(!done);

    return 0;
}


#include <stdio.h>

int main() {
    int ms, ps, np, psize[10];

    printf("Enter Memory Size: ");
    scanf("%d", &ms);

    printf("Enter Page Size: ");
    scanf("%d", &ps);

    printf("Enter Number of Processes: ");
    scanf("%d", &np);

    for(int i = 0; i < np; i++) {
        printf("Enter size of Process %d: ", i + 1);
        scanf("%d", &psize[i]);
    }

    printf("\nProcess\tPages Required\n");

    for(int i = 0; i < np; i++) {
        int pages = (psize[i] + ps - 1) / ps; // Ceiling division
        printf("P%d\t%d\n", i + 1, pages);
    }

    printf("\nTotal Frames in Memory = %d\n", ms / ps);

    return 0;
}






#include <stdio.h>

int main() {
    int n, at[10], bt[10], pr[10], done[10]={0};
    int time=0, count=0, min, p;

    printf("No of processes: ");
    scanf("%d",&n);

    for(int i=0;i<n;i++)
        scanf("%d%d%d",&at[i],&bt[i],&pr[i]);

    while(count<n){
        min=999; p=-1;

        for(int i=0;i<n;i++)
            if(at[i]<=time && !done[i] && pr[i]<min){
                min=pr[i];
                p=i;
            }

        if(p==-1) time++;
        else{
            time+=bt[p];
            printf("P%d completed at %d\n",p+1,time);
            done[p]=1;
            count++;
        }
    }
    return 0;
}
